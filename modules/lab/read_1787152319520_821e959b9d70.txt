   425→// Track Card Component
   426→function TrackCard({
   427→  track,
   428→  index,
   429→  isSelected,
   430→  onSelect,
   431→  onRemove,
   432→  onVolumeChange,
   433→  onPanChange,
   434→  onToggleMute,
   435→  onToggleSolo,
   436→  onToggleArm,
   437→  theme,
   438→}: {
   439→  track: {
   440→    id: string
   441→    name: string
   442→    color: string
   443→    volume: number
   444→    pan: number
   445→    muted: boolean
   446→    solo: boolean
   447→    armed: boolean
   448→  }
   449→  index: number
   450→  isSelected: boolean
   451→  onSelect: () => void
   452→  onRemove: () => void
   453→  onVolumeChange: (volume: number) => void
   454→  onPanChange: (pan: number) => void
   455→  onToggleMute: () => void
   456→  onToggleSolo: () => void
   457→  onToggleArm: () => void
   458→  theme: { primaryColor: string; textColor: string; surfaceColor: string; gridColor: string }
   459→}) {
   460→  return (
   461→    <div
   462→      onClick={onSelect}
   463→      className={`p-3 rounded-lg cursor-pointer transition-all ${
   464→        isSelected ? 'ring-2 ring-offset-1' : ''
   465→      }`}
   466→      style={{
   467→        backgroundColor: isSelected ? `${track.color}20` : theme.surfaceColor,
   468→        borderColor: isSelected ? track.color : 'transparent',
   469→        '--tw-ring-color': track.color,
   470→      } as React.CSSProperties}
   471→    >
   472→      <div className="flex items-center justify-between mb-2">
   473→        <div className="flex items-center gap-2">
   474→          <div
   475→            className="w-3 h-3 rounded-full"
   476→            style={{ backgroundColor: track.color }}
   477→          />
   478→          <span className="font-medium text-sm truncate" style={{ color: theme.textColor }}>
   479→            {track.name}
   480→          </span>
   481→        </div>
   482→        <button
   483→          onClick={(e) => { e.stopPropagation(); onRemove() }}
   484→          className="p-1 rounded hover:bg-red-100 text-gray-400 hover:text-red-500"
   485→        >
   486→          <TrashIcon />
   487→        </button>
   488→      </div>
   489→
   490→      <div className="space-y-2">

   496→            min="0"
   497→            max="1"
   498→            step="0.01"
   499→            value={track.volume}
   500→            onChange={(e) => onVolumeChange(Number(e.target.value))}
   501→            className="flex-1 h-1 accent-current"
   502→            style={{ color: track.color }}
   503→          />

   513→            type="range"
   514→            min="-1"
   515→            max="1"

   520→            style={{ color: track.color }}
   521→          />
   522→          <span className="text-xs w-4 text-center" style={{ color: theme.textColor }}>R</span>
   523→        </div>
   524→
   525→        {/* Controls Row */}
   526→        <div className="flex items-center justify-between pt-1">

   528→            <button
   529→              onClick={(e) => { e.stopPropagation(); onToggleMute() }}

   664→// Timeline Grid Component
   665→function TimelineGrid({
   666→  bpm,
   667→  timeSignature,
   668→  zoomLevel,
   669→  currentTime,
   670→  loopStart,
   671→  loopEnd,
   672→  loopEnabled,
   673→  tracks,
   674→  onTimeClick,
   675→  theme,
   676→}: {
   677→  bpm: number
   678→  timeSignature: { beatsPerMeasure: number; beatUnit: number }
   679→  zoomLevel: number
   680→  currentTime: number
   681→  loopStart: number
   682→  loopEnd: number
   683→  loopEnabled: boolean
   684→  tracks: Array<{ id: string; name: string; color: string; clips: Array<{ id: string; startTime: number; duration: number }> }>
   685→  onTimeClick: (time: number) => void
   686→  theme: { gridColor: string; surfaceColor: string; primaryColor: string }
   687→}) {
   688→  const duration = 120 // Show 2 minutes of timeline
   689→  const pixelsPerSecond = zoomLevel
   690→  const totalWidth = duration * pixelsPerSecond
   691→
   692→  // Generate time markers
   693→  const markers = []
   694→  const secondsPerBeat = 60 / bpm
   695→  for (let t = 0; t <= duration; t += secondsPerBeat) {
   696→    markers.push(t)
   697→  }
   698→
   699→  return (
   700→    <div className="absolute inset-0 overflow-auto">
   701→      <div style={{ width: totalWidth, minHeight: '100%' }} className="relative">
   702→        {/* Time Ruler */}
   703→        <div
   704→          className="sticky top-0 h-8 border-b flex items-end z-10"
   705→          style={{
   706→            backgroundColor: theme.surfaceColor,
   707→            borderColor: theme.gridColor,
   708→          }}
   709→        >
   710→          {markers.map((t) => (
   711→            <div
   712→              key={t}
   713→              className="absolute bottom-0 text-xs text-gray-500"
   714→              style={{ left: t * pixelsPerSecond }}
   715→            >
   716→              <div className="w-px h-2 bg-gray-300 mb-1 mx-auto" />
   717→              <span className="whitespace-nowrap transform -translate-x-1/2 block text-center">
   718→                {formatTimeDisplay(t)}
   719→              </span>
   720→            </div>
   721→          ))}
   722→        </div>
   723→
   724→        {/* Loop Region */}
   725→        {loopEnabled && (
   726→          <div
   727→            className="absolute top-8 h-full opacity-30 pointer-events-none"
   728→            style={{
   729→              left: loopStart * pixelsPerSecond,
   730→              width: (loopEnd - loopStart) * pixelsPerSecond,
   731→              backgroundColor: theme.primaryColor,
   732→            }}
   733→          />
   734→        )}
   735→
   736→        {/* Track Lanes */}
   737→        <div className="pt-8">
   738→          {tracks.map((track) => (
   739→            <div
   740→              key={track.id}
   741→              className="h-20 border-b relative"
   742→              style={{ borderColor: theme.gridColor }}
   743→            >
   744→              {/* Clips */}
   745→              {track.clips.map((clip) => (
   746→                <div
   747→                  key={clip.id}
   748→                  className="absolute top-2 h-16 rounded-md shadow-sm cursor-pointer hover:shadow-md transition-shadow"
   749→                  style={{
   750→                    left: clip.startTime * pixelsPerSecond,
   751→                    width: Math.max(clip.duration * pixelsPerSecond, 20),
   752→                    backgroundColor: track.color,
   753→                    opacity: 0.85,
   754→                  }}
   755→                >
   756→                  <div className="px-2 py-1 h-full flex flex-col justify-between">
   757→                    <span className="text-xs font-medium text-white truncate">
   758→                      {clip.name || 'Clip'}
   759→                    </span>
   760→                    <span className="text-xs text-white/70">
   761→                      {formatTimeDisplay(clip.duration)}
   762→                    </span>
   763→                  </div>
   764→                </div>
   765→              ))}
   766→            </div>
   767→          ))}
   768→        </div>
   769→
   770→        {/* Click to seek */}
   771→        <div
   772→          className="absolute inset-0 pt-8 cursor-crosshair"
   773→          onClick={(e) => {
   774→            const rect = e.currentTarget.getBoundingClientRect()
   775→            const x = e.clientX - rect.left
   776→            const time = x / pixelsPerSecond
   777→            onTimeClick(Math.max(0, time))
   778→          }}
   779→        />
   780→      </div>
   781→    </div>
   782→  )
   783→}
